4. 实现拖拉拽功能 ​

javascript
function enableDrag(element) {
    let isDragging = false;
    let currentX;
    let currentY;
    let initialX;
    let initialY;

    element.addEventListener('mousedown', dragStart);
    document.addEventListener('mousemove', drag);
    document.addEventListener('mouseup', dragEnd);

    function dragStart(e) {
        initialX = e.clientX - currentX;
        initialY = e.clientY - currentY;
        isDragging = true;
    }

    function drag(e) {
        if (!isDragging) return;
        e.preventDefault();
        currentX = e.clientX - initialX;
        currentY = e.clientY - initialY;
        element.style.transform = `translate(${currentX}px, ${currentY}px)`;
    }

    function dragEnd() {
        isDragging = false;
    }
}

function shuffle(arr) {
    for (let i = arr.length - 1; i > 0; i--) {

7. 对象深拷贝 ​

javascript
// 方法1：使用JSON
function deepClone1(obj) {
    return JSON.parse(JSON.stringify(obj));
}

// 方法2：递归实现
function deepClone2(obj, hash = new WeakMap()) {
    if (obj === null || typeof obj !== 'object') return obj;
    if (obj instanceof Date) return new Date(obj);
    if (obj instanceof RegExp) return new RegExp(obj);
    
    // 处理循环引用
    if (hash.has(obj)) return hash.get(obj);
    
    const cloneObj = Array.isArray(obj) ? [] : {};
    hash.set(obj, cloneObj);
    
    for (let key in obj) {
        if (obj.hasOwnProperty(key)) {
            cloneObj[key] = deepClone2(obj[key], hash);
        }
    }
    return cloneObj;
}

8. 实现一个柯里化函数add ​

javascript
function curry(fn) {
    return function curried(...args) {
        if (args.length >= fn.length) {
            return fn.apply(this, args);
        }
        return function(...args2) {
            return curried.apply(this, args.concat(args2));
        }
    };
}

// 实现add(1)(2)(3)
function add(...args) {
    let sum = args.reduce((acc, cur) => acc + cur, 0);
    
    function innerAdd(...newArgs) {
        sum += newArgs.reduce((acc, cur) => acc + cur, 0);
        return innerAdd;
    }
    
    innerAdd.toString = () => sum;
    return innerAdd;
}

// 使用示例
console.log(add(1)(2)(3).toString()); // 6

10. 实现一个防抖函数 ​

javascript
function debounce(fn, delay) {
    let timer = null;
    return function (...args) {
        if (timer) clearTimeout(timer);
        timer = setTimeout(() => {
            fn.apply(this, args);
            timer = null;
        }, delay);
    };
}

// 使用示例
const debouncedFn = debounce(() => console.log('执行'), 1000);

11. 实现一个节流函数 ​

javascript
function throttle(fn, delay) {
    let lastTime = 0;
    return function (...args) {
        const now = Date.now();
        if (now - lastTime >= delay) {
            fn.apply(this, args);
            lastTime = now;
        }
    };
}

// 使用示例
const throttledFn = throttle(() => console.log('执行'), 1000);

12. 实现一个方法，能上传多张图片，保持单次n张上传，如果有1张上传成功，然后就补上一张，直到全部上传成功 ​

javascript
class ImageUploader {
    constructor(maxConcurrent = 3) {
        this.maxConcurrent = maxConcurrent;
        this.queue = [];
        this.running = 0;
    }

    async upload(images) {
        // 将所有图片加入队列
        this.queue.push(...images);
        this.processQueue();
    }

    async processQueue() {
        while (this.queue.length && this.running < this.maxConcurrent) {
            const image = this.queue.shift();
            this.running++;
            
            try {
                await this.uploadSingle(image);
            } finally {
                this.running--;
                this.processQueue();
            }
        }
    }

    async uploadSingle(image) {
        // 实际上传逻辑
        return new Promise((resolve, reject) => {
            const formData = new FormData();
            formData.append('image', image);
            
            fetch('/upload', {
                method: 'POST',
                body: formData
            })
            .then(resolve)
            .catch(reject);
        });
    }
}

14. 实现一个 once 函数 ​

javascript
function once(fn) {
    let called = false;
    let result;
    
    return function (...args) {
        if (!called) {
            called = true;
            result = fn.apply(this, args);
            return result;
        }
        return result;
    };
}

// 使用示例
const onceFunc = once((x) => {
    console.log('只执行一次:', x);
    return x;
});

onceFunc(1); // 只执行一次: 1
onceFunc(2); // 不会执行，返回第一次的结果

16. 将原生的 ajax 封装成 promise ​

function ajax(options) {
    return new Promise((resolve, reject) => {
        const xhr = new XMLHttpRequest();
        
        xhr.open(options.method || 'GET', options.url);
        
        if (options.headers) {
            Object.keys(options.headers).forEach(key => {
                xhr.setRequestHeader(key, options.headers[key]);
            });
        }
        
        xhr.onload = function() {
            if (xhr.status >= 200 && xhr.status < 300) {
                resolve(xhr.response);
            } else {
                reject(new Error(xhr.statusText));
            }
        };
        
        xhr.onerror = function() {
            reject(new Error('Network Error'));
        };
        
        xhr.send(options.data);
    });
}

// 使用示例
ajax({
    url: '/api/data',
    method: 'POST',
    headers: {
        'Content-Type': 'application/json'
    },
    data: JSON.stringify({ key: 'value' })
}).then(response => {
    console.log(response);
}).catch(error => {
    console.error(error);
});

19. 实现响应式数据 + 依赖收集 ​

javascript
// 存储当前正在执行的副作用函数
let activeEffect = null;

// 副作用函数注册器
function effect(fn) {
    activeEffect = fn;
    fn(); // 立即执行一次，进行依赖收集
    activeEffect = null;
}

// 依赖管理类
class Dep {
    constructor() {
        this.subscribers = new Set(); // 使用 Set 存储依赖
    }

    // 收集依赖
    depend() {
        if (activeEffect) {
            this.subscribers.add(activeEffect);
        }
    }

    // 触发依赖
    notify() {
        this.subscribers.forEach(effect => effect());
    }
}

// 用于存储每个对象的依赖关系
const targetMap = new WeakMap();

// 获取某个对象某个属性的依赖收集器
function getDep(target, key) {
    let depsMap = targetMap.get(target);
    if (!depsMap) {
        depsMap = new Map();
        targetMap.set(target, depsMap);
    }
    
    let dep = depsMap.get(key);
    if (!dep) {
        dep = new Dep();
        depsMap.set(key, dep);
    }
    
    return dep;
}

// 创建响应式对象
function reactive(target) {
    return new Proxy(target, {
        // 拦截读取操作
        get(target, key, receiver) {
            const dep = getDep(target, key);
            dep.depend(); // 收集依赖
            return Reflect.get(target, key, receiver);
        },
        
        // 拦截设置操作
        set(target, key, value, receiver) {
            const result = Reflect.set(target, key, value, receiver);
            const dep = getDep(target, key);
            dep.notify(); // 触发依赖
            return result;
        }
    });
}

// 使用示例
const state = reactive({
    count: 0,
    message: 'Hello'
});

// 创建一个响应式效果
effect(() => {
    console.log('Count changed:', state.count);
});

// 创建另一个响应式效果
effect(() => {
    console.log('Message changed:', state.message);
});

// 修改值会触发相应的效果
state.count++; // 输出: Count changed: 1
state.message = 'World'; // 输出: Message changed: World

// 支持嵌套对象
const nested = reactive({
    user: {
        name: 'John',
        age: 25
    }
});

effect(() => {
    console.log('User name changed:', nested.user.name);
});

nested.user.name = 'Jane'; // 输出: User name changed: Jane

20. 手动实现一个 instanceOf 方法 ​

javascript
function myInstanceOf(obj, constructor) {
    // 处理基本类型和null
    if (obj === null || typeof obj !== 'object') {
        return false;
    }

    let proto = Object.getPrototypeOf(obj);
    
    while (proto !== null) {
        if (proto === constructor.prototype) {
            return true;
        }
        proto = Object.getPrototypeOf(proto);
    }
    
    return false;
}

// 使用示例
class Animal {}
class Dog extends Animal {}

const dog = new Dog();
console.log(myInstanceOf(dog, Dog));      // true
console.log(myInstanceOf(dog, Animal));   // true
console.log(myInstanceOf(dog, Object));   // true
console.log(myInstanceOf(dog, Array));    // false

21. 还原一颗树 ​

// 假设输入数据格式为扁平化的节点列表，每个节点包含id和parentId
function restoreTree(nodes) {
    const nodeMap = new Map();
    let root = null;

    // 首先创建一个节点映射
    nodes.forEach(node => {
        nodeMap.set(node.id, {
            ...node,
            children: []
        });
    });

    // 构建树结构
    nodes.forEach(node => {
        const currentNode = nodeMap.get(node.id);
        if (node.parentId === null) {
            root = currentNode;
        } else {
            const parentNode = nodeMap.get(node.parentId);
            if (parentNode) {
                parentNode.children.push(currentNode);
            }
        }
    });

    return root;
}

// 使用示例
const nodes = [
    { id: 1, parentId: null, name: 'root' },
    { id: 2, parentId: 1, name: 'child1' },
    { id: 3, parentId: 1, name: 'child2' },
    { id: 4, parentId: 2, name: 'grandchild' }
];

const tree = restoreTree(nodes);

22. 实现 b 函数，使得 b.then 中能打印出 cb 的参数值 ​

function b(cb) {
    return {
        then: function(resolve) {
            // 包装回调函数，使其能够捕获参数
            const wrappedCb = (...args) => {
                const result = cb(...args);
                resolve(result);
            };
            
            // 返回包装后的回调函数
            return wrappedCb;
        }
    };
}

// 使用示例
function callback(x) {
    return x * 2;
}

const wrapped = b(callback);
wrapped.then(result => {
    console.log(result); // 将打印回调函数的返回值
});

// 调用包装后的函数
wrapped(5); // 会在then中打印出10
